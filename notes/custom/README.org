** Project Config
*** Source Config

#+NAME: conf-custom
#+BEGIN_SRC bash
  [ ./etc/custom/custom.conf ] || sed 's/^ *//' >> ./etc/custom/custom.conf <<-EOF
  # project customization config
  set -a
  . ./etc/custom/custom-source.conf  
  . ./etc/custom/custom-target.conf  
  set -a
  EOF
#+END_SRC

#+NAME: custom-source.conf
#+BEGIN_SRC bash
  [ ./etc/custom/custom-source.conf ] || sed 's/^ *//' >> ./etc/custom/custom-source.conf <<-EOF
  # project source consts
  CUST_S_PROJECT_NAME='dve-sample-r'
  CUST_S_PACKAGE_NAME='dvesimpler'
  CUST_S_REPO_PATH='ub-dems-public/ds-labs'
  CUST_S_REPO_HOST='https://gitlab.com/'
  EOF
#+END_SRC
