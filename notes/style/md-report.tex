% Page Geometry
% ------------------------------------------------------------------------
\usepackage[
  top=2.5cm,
  bottom=2.5cm,
  left=2.2cm,
  right=2.2cm,
  headheight=14pt,
  headsep=0.8cm
]{geometry}

% ------------------------------------------------------------------------
% Paragraph & Line Spacing
% ------------------------------------------------------------------------
% Vertical space between paragraphs
\setlength{\parskip}{0.8em}

% No paragraph indentation (space-separated paragraphs instead)
\setlength{\parindent}{0pt}

% Line spacing (1.15 = slightly more than single-spaced)
\linespread{1.15}

% ------------------------------------------------------------------------
% Page Layout: Headers & Footers
% ------------------------------------------------------------------------
\usepackage{fancyhdr}
\pagestyle{fancy}

% Clear default header/footer
\fancyhead{}
\fancyfoot{}

% Left page (even): section name on right, chapter name on left
% Right page (odd): section name on right, chapter name on left
%\fancyhead[LE,RO]{\nouppercase{\slshape \rightmark}}
%\fancyhead[LO,RE]{\nouppercase{\slshape \leftmark}}

\fancyhead[L]{\nouppercase{\small\slshape \rightmark}}
\fancyhead[R]{\nouppercase{\small\slshape \leftmark}}

% Centered page number in footer
\fancyfoot[C]{\small\thepage}

% Header rule thickness
\renewcommand{\headrulewidth}{0.4pt}

% ------------------------------------------------------------------------
% Section Heading Customization
% ------------------------------------------------------------------------
\usepackage{titlesec}

% Chapter format: display style with "Chapter N" label
\titleformat{\chapter}[display]%
  {\normalfont\huge\sffamily\bfseries}%
  {\chaptertitlename\ \thechapter}%
  {20pt}%
  {\Huge}

% Chapter spacing: no extra space before, 40pt after
\titlespacing*{\chapter}{0pt}{0pt}{40pt}

% Section formats: large bold with number
\titleformat{\section}%
  {\sffamily\Large\bfseries}%
  {\thesection}%
  {0.5em}%
  {}

\titleformat{\subsection}%
  {\sffamily\large\bfseries}%
  {\thesubsection}%
  {0.5em}%
  {}

\titleformat{\subsubsection}%
  {\sffamily\normalsize\bfseries}%
  {\thesubsubsection}%
  {0.5em}%
  {}
